from sqlalchemy.orm import relationship

class UserFollow(Base):

### 阶段二：后端服务层开发（优先级：高）

#### 任务4：创建关注服务层FollowService

**文件位置：** `backend/app/services/follow_service.py`

**任务内容：**
- 创建 `FollowService` 类（参考 LikeService）
- 实现方法：
  - `toggle_follow`：关注/取消关注（直接删除记录）
  - `is_following`：检查关注状态

**代码结构（参考 LikeService）：**
```python
"""
 * @file            backend/app/services/follow_service.py
 * @description     关注服务层
 * @author          圆运阁古易文化 <gordon_cao@qq.com>
 * @createTime      2026-03-24 19:23:00
 * @lastModified    2026-03-24 19:23:00
 * Copyright © All rights reserved
"""

from typing import Optional, Dict, Any
from sqlalchemy.orm import Session
from app.models.user_follow import UserFollow

class FollowService:
    """关注服务类"""
    
    def __init__(self, db: Session):
        self.db = db
    
    def toggle_follow(self, follower_id: int, followed_id: int) -> Dict[str, Any]:
        """
        关注/取消关注（直接删除记录）
        
        Args:
            follower_id: 关注者用户ID
            followed_id: 被关注者用户ID
        
        Returns:
            操作结果（is_followed）
        """
        # 防止关注自己
        if follower_id == followed_id:
            raise ValueError("不能关注自己")
        
        # 查询现有关系
        follow = self.db.query(UserFollow).filter(
            UserFollow.follower_id == follower_id,
            UserFollow.followed_id == followed_id
        ).first()
        
        if follow:
            # 已关注 -> 取消关注（直接删除）
            self.db.delete(follow)
            self.db.commit()
            
            return {
                "is_followed": False
            }
        else:
            # 未关注 -> 关注
            new_follow = UserFollow(
                follower_id=follower_id,
                followed_id=followed_id
            )
            self.db.add(new_follow)
            self.db.commit()
            
            return {
                "is_followed": True
            }
    
    def is_following(self, follower_id: int, followed_id: int) -> bool:
        """
        获取关注状态
        
        Args:
            follower_id: 关注者用户ID
            followed_id: 被关注者用户ID
        
        Returns:
            是否已关注
        """
        follow = self.db.query(UserFollow).filter(
            UserFollow.follower_id == follower_id,
            UserFollow.followed_id == followed_id
        ).first()
        
        return follow is not None
```
