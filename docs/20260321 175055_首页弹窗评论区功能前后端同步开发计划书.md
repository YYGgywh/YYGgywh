```python
# backend/app/api/comment.py

class CommentResponse(BaseModel):
    id: int
    content: str
    create_time: int
    user_id: int
    user_nickname: str
    user_avatar: str
    is_author: bool
    is_public: bool

class ListCommentResponse(BaseModel):
    code: int = 200
    msg: str = "查询成功"
    data: dict  # 改为 dict，包含 list 和 total

@router.get("/list", response_model=ListCommentResponse)
async def list_comment(
    pan_record_id: int = Query(..., description="排盘记录ID"),
    page: int = Query(default=1, ge=1, description="页码"),
    size: int = Query(default=10, ge=1, le=100, description="每页数量"),
    current_user: User = Depends(get_current_user_from_payload),
    db: Session = Depends(get_db),
    req: Request = Depends(rate_limit_dependency)
):
    """查询评论列表"""
    # 计算偏移量
    offset = (page - 1) * size
    
    # 查询总数
    total = db.query(Comment).filter(
        Comment.pan_record_id == pan_record_id,
        Comment.is_visible == 1,
        Comment.deleted_at.is_(None)
    ).count()
    
    # 查询评论（关联用户信息）
    comments = db.query(Comment, User).join(
        User, Comment.user_id == User.id
    ).filter(
        Comment.pan_record_id == pan_record_id,
        Comment.is_visible == 1,
        Comment.deleted_at.is_(None)
    ).order_by(Comment.create_time.desc()).offset(offset).limit(size).all()
    
    # 获取排盘作者 ID
    pan_record = db.query(PanRecord).filter(PanRecord.id == pan_record_id).first()
    author_id = pan_record.user_id if pan_record else None
    
    # 构建响应数据
    data = []
    for comment, user in comments:
        data.append({
            "id": comment.id,
            "content": comment.content,
            "create_time": comment.create_time,
            "user_id": comment.user_id,
            "user_nickname": user.nickname or f"用户{user.id}",
            "user_avatar": user.avatar or "",
            "is_author": comment.user_id == author_id,
            "is_public": comment.is_public
        })
    
    return ListCommentResponse(data={
        "list": data,
        "total": total,
        "page": page,
        "size": size
    })
```

```python
import os
import sqlite3

BASE_DIR = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
DB_PATH = os.path.join(BASE_DIR, "db", "yyggywh.db")

def migrate():
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    
    try:
        # SQLite 不支持直接修改列类型，需要重建表
        cursor.execute("""
            CREATE TABLE comment_new (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                pan_record_id INTEGER NOT NULL,
                user_id INTEGER NOT NULL,
                content TEXT NOT NULL,
                create_time INTEGER DEFAULT CURRENT_TIMESTAMP,
                update_time INTEGER DEFAULT CURRENT_TIMESTAMP,
                is_public BOOLEAN DEFAULT 0,
                ext_info TEXT DEFAULT '{}',
                audit_status INTEGER DEFAULT 0,
                audit_time INTEGER,
                audit_user_id INTEGER,
                audit_remark TEXT,
                is_visible INTEGER DEFAULT 1,
                deleted_at INTEGER
            )
        """)
        
        # 复制数据
        cursor.execute("INSERT INTO comment_new SELECT * FROM comment")
        
        # 删除旧表
        cursor.execute("DROP TABLE comment")
        
        # 重命名新表
        cursor.execute("ALTER TABLE comment_new RENAME TO comment")
        
        # 重建索引
        cursor.execute("CREATE INDEX idx_comment_pan_user ON comment(pan_record_id, user_id)")
        cursor.execute("CREATE INDEX idx_comment_time ON comment(create_time)")
        
        conn.commit()
        print("迁移成功：comment.content 字段类型已改为 TEXT")
    except Exception as e:
        conn.rollback()
        print(f"迁移失败: {e}")
    finally:
        conn.close()

if __name__ == "__main__":
    migrate()
```

```javascript
import axios from 'axios';
import { getToken } from '../utils/storage';

// 创建 axios 实例
const api = axios.create({
  baseURL: process.env.REACT_APP_API_URL || 'http://localhost:8000/api/v1',
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
});

// 请求拦截器
api.interceptors.request.use(
  (config) => {
    const token = getToken();
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 响应拦截器
api.interceptors.response.use(
  (response) => {
    // 统一处理响应格式
    const { code, msg, data } = response.data;
    
    if (code !== 200) {
      // 业务错误
      return Promise.reject(new Error(msg || '请求失败'));
    }
    
    return data;
  },
  (error) => {
    // HTTP 错误处理
    if (error.response) {
      const { status, data } = error.response;
      
      switch (status) {
        case 401:
          // Token 过期，跳转登录
          window.location.href = '/login';
          break;
        case 403:
          console.error('无权限操作');
          break;
        case 429:
          console.error('请求过于频繁，请稍后再试');
          break;
        default:
          console.error(data?.msg || '服务器错误');
      }
    }
    
    return Promise.reject(error);
  }
);

export default api;
```

```python
from typing import List, Dict, Any
from sqlalchemy.orm import Session
from app.models.comment import Comment
from app.models.user import User
from app.models.pan_record import PanRecord

class CommentService:
    """评论服务类"""
    
    def __init__(self, db: Session):
        self.db = db
    
    def get_comment_list(
        self, 
        pan_record_id: int, 
        page: int = 1, 
        size: int = 10
    ) -> Dict[str, Any]:
        """获取评论列表"""
        offset = (page - 1) * size
        
        # 查询总数
        total = self.db.query(Comment).filter(
            Comment.pan_record_id == pan_record_id,
            Comment.is_visible == 1,
            Comment.deleted_at.is_(None)
        ).count()
        
        # 查询评论（关联用户信息）
        comments = self.db.query(Comment, User).join(
            User, Comment.user_id == User.id
        ).filter(
            Comment.pan_record_id == pan_record_id,
            Comment.is_visible == 1,
            Comment.deleted_at.is_(None)
        ).order_by(Comment.create_time.desc()).offset(offset).limit(size).all()
        
        # 获取排盘作者 ID
        pan_record = self.db.query(PanRecord).filter(
            PanRecord.id == pan_record_id
        ).first()
        author_id = pan_record.user_id if pan_record else None
        
        # 构建响应数据
        items = []
        for comment, user in comments:
            items.append({
                "id": comment.id,
                "content": comment.content,
                "create_time": comment.create_time,
                "user_id": comment.user_id,
                "user_nickname": user.nickname or f"用户{user.id}",
                "user_avatar": user.avatar or "",
                "is_author": comment.user_id == author_id,
                "is_public": comment.is_public
            })
        
        return {
            "list": items,
            "total": total,
            "page": page,
            "size": size
        }
```

```python
def increment_comment_count(self, pan_record_id: int):
    """增加评论计数"""
    record = self.db.query(PanRecord).filter(
        PanRecord.id == pan_record_id
    ).first()
    
    if record:
        record.comment_count = (record.comment_count or 0) + 1
        self.db.commit()

def decrement_comment_count(self, pan_record_id: int):
    """减少评论计数"""
    record = self.db.query(PanRecord).filter(
        PanRecord.id == pan_record_id
    ).first()
    
    if record and record.comment_count > 0:
        record.comment_count -= 1
        self.db.commit()
```

```python
    # 查询当前点赞数
    like_count = db.query(CommentLike).filter(
        CommentLike.comment_id == request.comment_id
    ).count()
    
    return ToggleCommentLikeResponse(data={
        "is_liked": is_liked,
        "like_count": like_count
    })
```
